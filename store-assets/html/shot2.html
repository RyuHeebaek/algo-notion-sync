17import java.util.Scanner;

public class Solution {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int T = sc.nextInt();

        for (int tc = 1; tc <= T; tc++) {
            int[] cnt = new int[101];
            for (int i = 0; i < 1000; i++) cnt[sc.nextInt()]++;

            int mode = 100;
            for (int s = 99; s >= 0; s--) if (cnt[s] > cnt[mode]) mode = s;
            System.out.println("#" + tc + " " + mode);
        }
    }
}
